# Run these in the server console or in-game as OP after plugins load.
# Server must be stopped for some file copies — see README.

# --- Worlds ---
/mv create lobby NORMAL
/mv create creative NORMAL
/mv create survival NORMAL
/mv create pvp FLAT

/mv modify set gamemode ADVENTURE lobby
/mv modify set gamemode CREATIVE creative
/mv modify set gamemode SURVIVAL survival
/mv modify set gamemode SURVIVAL pvp
/mv modify set difficulty PEACEFUL creative
/mv modify set difficulty NORMAL survival

/mv gamerule creative doMobSpawning false
/mv gamerule pvp keepInventory true
/mv gamerule lobby doDaylightCycle false
/mv gamerule lobby doMobSpawning false

/mv set spawn lobby
/setspawn

/mv weather lobby sun
/mv weather lobby sun permanent

# --- Multiverse-Inventories ---
/mvinv enable
/mv reload

# --- Essentials aliases ---
/setalias hub /spawn
/setalias lobby /spawn
